class Solution {
    public void sortStack(Stack<Integer> st) {
        if(st.isEmpty()){
            return;
        }
        
        int element=st.pop();
        sortStack(st);
        insertsort(st , element);
    }
    public void insertsort(Stack<Integer>st ,int element){
        if(st.isEmpty() || st.peek() <= element){
            st.push(element);
            return;
        }
        int peek = st.pop();
        insertsort(st , element);
        st.push(peek);
    }
}
