/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<Integer> Sametree(List<Integer> result, TreeNode p) {
        if (p == null) {
            result.add(null);
            return result;
        }
        Sametree(result, p.left);
        Sametree(result, p.right);
        result.add(p.val);
        return result;
    }

    public boolean isSameTree(TreeNode p, TreeNode q) {
        List<Integer> result1 = new ArrayList<>();
        List<Integer> result2 = new ArrayList<>();
        List<Integer> ptree = Sametree(result1, p);
        List<Integer> qtree = Sametree(result2, q);
        if (ptree.equals(qtree)) {
            return true;
        } else {
            return false;
        }
    }
}
