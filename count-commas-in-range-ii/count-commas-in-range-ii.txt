class Solution {
    public long countCommas(long n) {

        long lower = 1000;
        int commas = 1;
        long result = 0;
        while (lower <= n) {
          long  upper = lower * 1000 - 1;
            if (upper > n) {
                upper = n;
            }
            long num = (upper - lower + 1);
            result += num * commas;
            lower = lower * 1000;
            commas++;
        }
        return result;
    }
}
