class Solution {
    public int[][] cyclicShift(int n, int[][] grid, int[] row, int[] col) {
        int ans[][] = new int[n][n];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {

                
                int r = (i + col[j]) % n;

           
                int c = (j + row[r]) % n;

                ans[i][j] = grid[r][c];
            }
        }

        return ans;
    }
}
