class Solution {
    public List<Integer> targetIndices(int[] nums, int target) {
        int smallerthan=0;
        int count=0;
        for(int i=0; i<nums.length; i++){
            if(nums[i]<target){
                smallerthan++;
            }
            if(nums[i]==target){
                count++;
            }
        }
        List<Integer> res= new ArrayList<>();
        for(int i=smallerthan; i<smallerthan+count; i++){
            res.add(i);
        }
        return res;
    }
}
