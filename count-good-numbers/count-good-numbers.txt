class Solution {
    public static int countGoodNumbers(long N) {
        long MOD =1000000007;
        long odd =N / 2;
        long even = N - odd;
        long ans =(power(5 ,even ,MOD)*power(4 ,odd,MOD))% MOD;
        return (int) ans;
    }
    public static long power(long base , long exp,long MOD){
        long res = 1;
        while(exp >0){
            if(exp % 2 == 1){
                res = (res * base) % MOD;
                exp = exp -1;
            }
            else{
                base = (base * base) % MOD;
                exp = exp / 2;
        }
    }
    return res;
}
}
