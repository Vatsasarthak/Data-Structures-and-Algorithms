class Solution {
    public int minMergedInteger(int[] nums) {
        int minMerged = Integer.MAX_VALUE;
        
        for (int i = 0; i < nums.length; i += 2) {
            int a = nums[i];
            int b = nums[i + 1];
            
            int multiplier = 10;
            if (b >= 1000) {
                multiplier = 10000;
            } else if (b >= 100) {
                multiplier = 1000;
            } else if (b >= 10) {
                multiplier = 100;
            }
            
            int merged = a * multiplier + b;
            
            if (merged < minMerged) {
                minMerged = merged;
            }
        }
        
        return minMerged;
    }
}
