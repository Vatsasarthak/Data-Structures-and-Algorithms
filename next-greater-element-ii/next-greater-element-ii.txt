class Solution {
    public int[] nextGreaterElements(int[] nums) {
        int[] res =new int[nums.length];
        Stack<Integer> st = new Stack<>();
        for (int i = 0; i < 2 * n; i++) {
            int currentVal = nums[i % n];
            while (!st.isEmpty() && currentVal > nums[st.peek()]) {
                int prevIndex = st.pop();
                res[prevIndex] = currentVal;
            }
            if (i < n) {
                st.push(i);
            }
        }
        
        return res;
    }
        int n= nums.length;
        Arrays.fill(res, -1);
}
