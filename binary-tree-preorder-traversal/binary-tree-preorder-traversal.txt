*     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<Integer> preorderTraversal(TreeNode root) {
        List<Integer> ans =new ArrayList<>();
        Stack<TreeNode> st=new Stack<TreeNode>();
        TreeNode node =root;
        while(true){
            if(node != null){
                st.push(node);
                ans.add(node.val);
                node =node.left;
            }
            else{
                if(st.isEmpty()){
                    break;
                }
                node =st.pop();
                node=node.right;
            }
        }
        return ans;
    }
}
