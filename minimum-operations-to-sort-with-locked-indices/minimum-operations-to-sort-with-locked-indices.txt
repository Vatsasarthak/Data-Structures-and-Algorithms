class Solution {
    public int minOperations(int[] A, int[] locked, int k) {
        int res = 0, mx = 0, n = A.length;
        for (int i = 0; i < n - 1; ++i) {
            mx = Math.max(mx, A[i]);
            if (mx > i + 1) {
                res += locked[i] + locked[i + 1];
                locked[i] = locked[i + 1] = 0;
            }
        }
        if (res > k) return -1;
        
        List<Integer> st = new ArrayList<>();
        for (int a : A) {
            int j = ~Collections.binarySearch(st, a);
            res += st.size() - j;
            st.add(j, a);
        }
        return res;
    }
}
