st.push(i);
        }
        return res;
    }

    public long sumSubarrayMaxs(int[] nums, int n) {
        int[] nextGreaterLeft = getNGL(nums, n);
        int[] nextGreaterRight = getNGR(nums, n);
        long sum = 0;
        
        for (int i = 0; i < n; i++) {
            long ls = i - nextGreaterLeft[i];
            long rs = nextGreaterRight[i] - i;
            
            long totalways = ls * rs;
            long totalsum = (long) nums[i] * totalways;
            
            sum = sum + totalsum;
        }
        
        return sum;
    }
    public long subArrayRanges(int[] nums) {
        int n=nums.length;
        long sum =sumSubarrayMaxs(nums , n) - sumSubarrayMins(nums , n );
        return sum;
}
