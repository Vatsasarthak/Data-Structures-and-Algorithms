        for(int i=0;i< asteroids.length;i++){
            boolean collide = true;
            while(!st.isEmpty() && st.peek() > 0 && asteroids[i] <0){
                if(st.peek() < -asteroids[i]){
                    st.pop();
                }
                else if(st.peek() == -asteroids[i]){
                    collide =false;
                    st.pop();
                    break;
                }
                else{
                    collide = false;
                    break;
                }
            }
            if(collide){
                st.push(asteroids[i]);
            }
        }
            int[] ans = new int[st.size()];
            for(int i=ans.length -1 ;i>= 0;i--){
                ans[i] = st.pop();
            }
    }
