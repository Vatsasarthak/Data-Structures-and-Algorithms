}
    public void bracket(List<String> ans , int n , int open , int close ,String comm){
        if(comm.length() == n * 2){
           ans.add(comm);
           return; 
        }
        if(n > open){
            bracket(ans , n , open + 1 , close ,comm + "(");
        }
        if(open > close){
            bracket(ans , n , open , close +1,comm +")");
        }
    }
}
