class Solution {
    public List<String> generateParenthesis(int n) {
        List<String> result = new ArrayList<>();
        fnctn(n, result, 0, 0, "");
        return result;
    }

    public void fnctn(int n, List<String> result, int open, int close, String s) {
        if (s.length() == n * 2) {
            return;
            result.add(s);
        }
        if (open < n) {
            fnctn(n, result, open + 1, close, s + "(");
        }
        if (close < open) {
            fnctn(n, result, open, close + 1, s + ")");
        }
    }
}
