        if(m > n) return false; 

        int[] pref = new int[n + 1], suf = new int[n + 1]; 

        int idx = 0; 
        for(int i = 0; i < n; i++) {
            pref[i + 1] = pref[i]; 
            if(idx < m) if(s.charAt(idx) == t.charAt(i)) {
                pref[i + 1]++; 
                idx++; 
            }
        }

        idx = m - 1;
        for (int j = n - 1; j >= 0; j--) {
            suf[j] = suf[j + 1];
            if (idx >= 0 && s.charAt(idx) == t.charAt(j)) {
                idx--;
                suf[j]++;
            }
        }

        for (int i = 0; i < n; i++) {
            if (pref[i] + suf[i] == m) return true;      
            if (pref[i] + suf[i + 1] >= m - 1) return true; 
        }
        
        return false; 
    }
