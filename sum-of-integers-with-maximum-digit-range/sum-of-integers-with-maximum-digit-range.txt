class Solution {
    public int maxDigitRange(int[] nums) {
        int maxRange = -1;
        int sum = 0;

        for (int num : nums) {
            int original = num;
            int max = 0;
            int min = 9;

            while (num > 0) {
                int digit = num % 10;
                max = Math.max(max, digit);
                min = Math.min(min, digit);
                num /= 10;
            }

            int range = max - min;

            if (range > maxRange) {
                maxRange = range;
                sum = original;
            } else if (range == maxRange) {
                sum += original;
            }
        }

        return sum;
    }
}
